include(FetchContent)

set(CMAKE_C_STANDARD 11)

get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)

# config/build_info.h does `#include "../../include/git_info.h"`. On the real
# ESP32 build, PlatformIO's pre_build.py (tools/build-scripts/) generates
# that file before every build; it's gitignored (include/git_info.h) and not
# part of this tree. sim-web has no PlatformIO pre-build step, so a fresh
# checkout building only sim-web would fail to find it. Generate a minimal
# placeholder here -- but only if the file doesn't already exist, so this
# never overwrites a real include/git_info.h produced by an actual
# PlatformIO/ESP32 build (and never touches .build_number, which only
# pre_build.py increments).
set(GIT_INFO_HEADER "${PROJECT_ROOT}/include/git_info.h")
if(NOT EXISTS "${GIT_INFO_HEADER}")
    file(WRITE "${GIT_INFO_HEADER}" "\
#pragma once

// Auto-generated placeholder for sim-web's standalone CMake build.
// Normally generated by tools/build-scripts/pre_build.py during the
// PlatformIO/ESP32 build; this file is gitignored (include/git_info.h).
// sim-web has no PlatformIO pre-build step, so sim-web/CMakeLists.txt
// creates this placeholder only if the file doesn't already exist -- it
// never overwrites a real one produced by an actual PlatformIO build.
// Safe to delete/regenerate locally at any time.

#define GIT_COMMIT_ID \"websim\"
#define GIT_BRANCH \"websim\"
#define BUILD_NUMBER 0
#define BUILD_TIMESTAMP \"unknown\"
")
endif()

# LVGL's own CMake target (built via FetchContent below) needs -sUSE_SDL=2 on
# its compile step too -- target_compile_options() on our own executable only
# affects our own translation units, not the lvgl library's. Setting it
# globally here (before FetchContent_MakeAvailable) covers both, same as
# sim-web/spike/CMakeLists.txt already proved.
add_compile_options("-sUSE_SDL=2")

FetchContent_Declare(
    lvgl
    GIT_REPOSITORY https://github.com/lvgl/lvgl.git
    GIT_TAG v9.5.0
    GIT_SHALLOW TRUE
)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(lvgl)

target_compile_definitions(lvgl PUBLIC SMART_GRIND_WEB_SIM)
# Pin LVGL's config to sim-web/lv_conf.h (firmware config + browser overrides) by
# absolute path, for the library and, through PUBLIC, every app source too.
# Relying on include order doesn't work: CMake marks this directory -isystem,
# which searches after the app's -I paths, so include/lv_conf.h would win there
# and the app and library would disagree on LVGL's configuration.
target_compile_definitions(lvgl PUBLIC "LV_CONF_PATH=\"${CMAKE_CURRENT_LIST_DIR}/lv_conf.h\"")
target_include_directories(lvgl PUBLIC "${CMAKE_CURRENT_LIST_DIR}")

# --- Shadow source tree for menu_screen.cpp/.h ---
# menu_screen.h #includes real ESP32-only headers via relative paths
# ("../../bluetooth/manager.h", "../../controllers/grind_controller.h") that
# cannot be intercepted with an extra -I flag (relative quoted includes
# resolve against the including file's own directory first, and the real
# files genuinely exist there). Instead, copy the two real, unmodified
# files into a shadow tree positioned so those same relative paths land on
# our stand-in headers instead - re-copied on every build (see the
# add_custom_command DEPENDS below), never manually maintained, so it can
# never silently drift from the real source. If a future edit to
# menu_screen.cpp/.h calls something not in the stand-ins, the build fails
# loudly at that point - see sim-web/README.md "If the menu build breaks".
set(SHADOW_ROOT "${CMAKE_CURRENT_BINARY_DIR}/shadow-src")
set(SHADOW_MENU_CPP "${SHADOW_ROOT}/ui/screens/menu_screen.cpp")
set(SHADOW_MENU_H "${SHADOW_ROOT}/ui/screens/menu_screen.h")
# autotune_screen.h needs the same treatment as menu_screen.h: its own
# "../../controllers/autotune_controller.h" would otherwise resolve first-try
# against the real src/controllers/, which drags in LittleFS, grinder.h and
# the whole grind path. From inside the shadow tree it lands on the stand-in.
set(SHADOW_AUTOTUNE_CPP "${SHADOW_ROOT}/ui/screens/autotune_screen.cpp")
set(SHADOW_AUTOTUNE_H "${SHADOW_ROOT}/ui/screens/autotune_screen.h")
set(SHADOW_HELPERS_H "${SHADOW_ROOT}/ui/ui_helpers.h")
set(SHADOW_EVENTBRIDGE_H "${SHADOW_ROOT}/ui/event_bridge_lvgl.h")
set(SHADOW_OVERLAY_H "${SHADOW_ROOT}/ui/components/blocking_overlay.h")
# The real nav bar (status_indicator_controller.cpp) includes "../ui_manager.h";
# from inside the shadow tree that lands on shadow-stubs/ui/ui_manager.h, which
# exposes just the two members it reads.
set(SHADOW_STATUS_CPP "${SHADOW_ROOT}/ui/controllers/status_indicator_controller.cpp")
set(SHADOW_STATUS_H "${SHADOW_ROOT}/ui/controllers/status_indicator_controller.h")
set(SHADOW_UI_MANAGER_H "${SHADOW_ROOT}/ui/ui_manager.h")

# config/*.h is NOT mirrored into SHADOW_ROOT (unlike ui_helpers.h/
# event_bridge_lvgl.h/blocking_overlay.h below). It's fully portable and
# menu_screen.cpp reaches it two ways in the same translation unit: directly
# via its own "../../config/constants.h", and transitively through the real,
# intentionally-unshadowed grinding_screen.h -> grinding_screen_arc.h/
# grinding_screen_chart.h, each of which has its own real
# "../../config/constants.h". A shadow copy would be a second, distinctly
# -pathed file with identical content - #pragma once guards per-file, not
# per-content, so both copies would get included and every enum/inline
# function in the config cluster would be reported as redefined. Instead,
# menu_screen.cpp/.h's own config includes fall through - same fallback
# mechanism as grinding_screen.h/gbw_icons.h below - to the one real copy.

# menu_screen.h/.cpp's OTHER "../../X" relative includes (bluetooth/manager.h,
# controllers/grind_controller.h, system/diagnostics_controller.h,
# system/statistics_manager.h, hardware/hardware_manager.h,
# logging/grind_logging.h)
# need the exact same first-try treatment as menu_screen.cpp/.h itself: a
# quoted "../../X" include resolves against the including file's own
# directory first, unconditionally - an extra -I directory can only be
# reached once that first try fails, and (verified empirically) the
# fallback search concatenates the *entire* relative spec, ".." segments
# included, onto each -I directory - so an -I directory has to sit at
# exactly the right depth for that concatenation to land on the real
# target, which "shadow-stubs" (only one level below sim-web/, not two)
# does not. The reliable fix - matching the pattern this file already uses
# successfully for ui_helpers.h/event_bridge_lvgl.h/blocking_overlay.h above
# - is to mirror the hand-written shadow-stubs/ content into these same
# first-try-resolved paths under SHADOW_ROOT, refreshed on every build
# alongside the menu_screen.cpp/.h copy. shadow-stubs/ itself remains the
# single hand-maintained source of truth (git-tracked); this is a read-only
# build-time copy of it. (hardware/WeightSensor.h is included here too
# because shadow-stubs/hardware/hardware_manager.h needs its sibling
# "WeightSensor.h" to resolve the same way once relocated.)
set(SHADOW_STUBS_DIR "${CMAKE_CURRENT_LIST_DIR}/shadow-stubs")
set(SHADOW_STUB_MIRROR_FILES
    bluetooth/manager.h
    controllers/grind_controller.h
    system/diagnostics_controller.h
    system/statistics_manager.h
    hardware/hardware_manager.h
    hardware/WeightSensor.h
    logging/grind_logging.h
    controllers/autotune_controller.h
)
set(SHADOW_STUB_MIRROR_OUTPUTS "")
set(SHADOW_STUB_MIRROR_DEPENDS "")
foreach(f ${SHADOW_STUB_MIRROR_FILES})
    list(APPEND SHADOW_STUB_MIRROR_OUTPUTS "${SHADOW_ROOT}/${f}")
    list(APPEND SHADOW_STUB_MIRROR_DEPENDS "${SHADOW_STUBS_DIR}/${f}")
endforeach()

add_custom_command(
    OUTPUT ${SHADOW_MENU_CPP} ${SHADOW_MENU_H} ${SHADOW_HELPERS_H}
           ${SHADOW_EVENTBRIDGE_H} ${SHADOW_OVERLAY_H}
           ${SHADOW_AUTOTUNE_CPP} ${SHADOW_AUTOTUNE_H}
           ${SHADOW_STATUS_CPP} ${SHADOW_STATUS_H} ${SHADOW_UI_MANAGER_H}
           ${SHADOW_STUB_MIRROR_OUTPUTS}
    COMMAND ${CMAKE_COMMAND} -E make_directory
            "${SHADOW_ROOT}/ui/screens" "${SHADOW_ROOT}/ui/components" "${SHADOW_ROOT}/ui/controllers"
            "${SHADOW_ROOT}/bluetooth" "${SHADOW_ROOT}/controllers" "${SHADOW_ROOT}/system"
            "${SHADOW_ROOT}/hardware" "${SHADOW_ROOT}/logging"
    COMMAND ${CMAKE_COMMAND} -E copy "${PROJECT_ROOT}/src/ui/screens/menu_screen.cpp" "${SHADOW_MENU_CPP}"
    COMMAND ${CMAKE_COMMAND} -E copy "${PROJECT_ROOT}/src/ui/screens/menu_screen.h" "${SHADOW_MENU_H}"
    COMMAND ${CMAKE_COMMAND} -E copy "${PROJECT_ROOT}/src/ui/screens/autotune_screen.cpp" "${SHADOW_AUTOTUNE_CPP}"
    COMMAND ${CMAKE_COMMAND} -E copy "${PROJECT_ROOT}/src/ui/screens/autotune_screen.h" "${SHADOW_AUTOTUNE_H}"
    COMMAND ${CMAKE_COMMAND} -E copy "${PROJECT_ROOT}/src/ui/controllers/status_indicator_controller.cpp" "${SHADOW_STATUS_CPP}"
    COMMAND ${CMAKE_COMMAND} -E copy "${PROJECT_ROOT}/src/ui/controllers/status_indicator_controller.h" "${SHADOW_STATUS_H}"
    COMMAND ${CMAKE_COMMAND} -E copy "${SHADOW_STUBS_DIR}/ui/ui_manager.h" "${SHADOW_UI_MANAGER_H}"
    COMMAND ${CMAKE_COMMAND} -E copy "${PROJECT_ROOT}/src/ui/ui_helpers.h" "${SHADOW_HELPERS_H}"
    COMMAND ${CMAKE_COMMAND} -E copy "${PROJECT_ROOT}/src/ui/event_bridge_lvgl.h" "${SHADOW_EVENTBRIDGE_H}"
    COMMAND ${CMAKE_COMMAND} -E copy "${PROJECT_ROOT}/src/ui/components/blocking_overlay.h" "${SHADOW_OVERLAY_H}"
    COMMAND ${CMAKE_COMMAND} -E copy "${SHADOW_STUBS_DIR}/bluetooth/manager.h" "${SHADOW_ROOT}/bluetooth/manager.h"
    COMMAND ${CMAKE_COMMAND} -E copy "${SHADOW_STUBS_DIR}/controllers/grind_controller.h" "${SHADOW_ROOT}/controllers/grind_controller.h"
    COMMAND ${CMAKE_COMMAND} -E copy "${SHADOW_STUBS_DIR}/system/diagnostics_controller.h" "${SHADOW_ROOT}/system/diagnostics_controller.h"
    COMMAND ${CMAKE_COMMAND} -E copy "${SHADOW_STUBS_DIR}/system/statistics_manager.h" "${SHADOW_ROOT}/system/statistics_manager.h"
    COMMAND ${CMAKE_COMMAND} -E copy "${SHADOW_STUBS_DIR}/hardware/hardware_manager.h" "${SHADOW_ROOT}/hardware/hardware_manager.h"
    COMMAND ${CMAKE_COMMAND} -E copy "${SHADOW_STUBS_DIR}/hardware/WeightSensor.h" "${SHADOW_ROOT}/hardware/WeightSensor.h"
    COMMAND ${CMAKE_COMMAND} -E copy "${SHADOW_STUBS_DIR}/logging/grind_logging.h" "${SHADOW_ROOT}/logging/grind_logging.h"
    COMMAND ${CMAKE_COMMAND} -E copy "${SHADOW_STUBS_DIR}/controllers/autotune_controller.h" "${SHADOW_ROOT}/controllers/autotune_controller.h"
    DEPENDS
        "${PROJECT_ROOT}/src/ui/screens/menu_screen.cpp"
        "${PROJECT_ROOT}/src/ui/screens/menu_screen.h"
        "${PROJECT_ROOT}/src/ui/screens/autotune_screen.cpp"
        "${PROJECT_ROOT}/src/ui/screens/autotune_screen.h"
        "${PROJECT_ROOT}/src/ui/controllers/status_indicator_controller.cpp"
        "${PROJECT_ROOT}/src/ui/controllers/status_indicator_controller.h"
        "${SHADOW_STUBS_DIR}/ui/ui_manager.h"
        "${PROJECT_ROOT}/src/ui/ui_helpers.h"
        "${PROJECT_ROOT}/src/ui/event_bridge_lvgl.h"
        "${PROJECT_ROOT}/src/ui/components/blocking_overlay.h"
        ${SHADOW_STUB_MIRROR_DEPENDS}
    COMMENT "Refreshing shadow copy of menu_screen (re-run whenever the real file changes)"
)
add_custom_target(shadow_menu_tree DEPENDS ${SHADOW_MENU_CPP} ${SHADOW_MENU_H} ${SHADOW_HELPERS_H} ${SHADOW_EVENTBRIDGE_H} ${SHADOW_OVERLAY_H} ${SHADOW_AUTOTUNE_CPP} ${SHADOW_AUTOTUNE_H} ${SHADOW_STATUS_CPP} ${SHADOW_STATUS_H} ${SHADOW_UI_MANAGER_H} ${SHADOW_STUB_MIRROR_OUTPUTS})

target_include_directories(smart-grind-web-sim PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/platform"
    # NOTE: this entry was reassessed as dead after Task 5 (which never wrote
    # a bare `#include "hardware/WeightSensor.h"` etc. from outside the
    # shadow tree) and briefly removed while implementing Task 6 - but Task 6
    # adds exactly that from main.cpp (`WeightSensor stub_weight_sensor;`
    # needs the full type, not menu_screen.h's forward declaration of it).
    # Without this entry, that spec's -I search hits
    # "${PROJECT_ROOT}/src/hardware/WeightSensor.h" (the real HX711-backed
    # driver, ESP32-only) before it would ever reach the mirrored
    # "${SHADOW_ROOT}/hardware/WeightSensor.h" below, because this entry sits
    # before "${PROJECT_ROOT}/src" in the search order and that one doesn't.
    # Confirmed empirically: removing it breaks main.cpp's build once Task 6's
    # WeightSensor global is added. Restored - this is live, not dead.
    "${CMAKE_CURRENT_LIST_DIR}/shadow-stubs"
    # shadow-stubs/ui/event_bridge_lvgl.cpp's own `#include "event_bridge_lvgl.h"`
    # (bare, no relative prefix) isn't next to itself in shadow-stubs/ui/ - the
    # real header only exists as the Step-1 copy at SHADOW_ROOT/ui/ - so this
    # narrow entry (not the whole of SHADOW_ROOT - see below) resolves it.
    "${SHADOW_ROOT}/ui"
    # menu_screen.cpp's `#include "grinding_screen.h"` (bare) and
    # `#include "../icons/gbw_icons.h"` (one level up) both intentionally fall
    # through to the real, unshadowed src/ui/screens/ tree (see Step 11's
    # resolution-order note) - this entry is what makes that fallback land.
    "${PROJECT_ROOT}/src/ui/screens"
    # The shadow copy of ui_helpers.h (SHADOW_ROOT/ui/ui_helpers.h, one level
    # below SHADOW_ROOT, mirroring its real position one level below
    # src/ui/) has its own "../config/constants.h" - this is the matching
    # one-level-below-src fallback entry for that, landing on the one real
    # copy (same reasoning as the config/* note above: no separate shadow
    # copy of config exists to redirect to).
    "${PROJECT_ROOT}/src/ui"
    # main.cpp's own top-level #include "config/constants.h" doesn't resolve
    # next to main.cpp (sim-web/), so it falls through this search list -
    # it needs to land here, on the one real copy.
    "${PROJECT_ROOT}/src"
    "${SHADOW_ROOT}"
    "${PROJECT_ROOT}/include"
    "${lvgl_SOURCE_DIR}/src"
)
